cmake_minimum_required(VERSION 3.20)
project(CHttpPrivateServiceHostStagedConsumer LANGUAGES C CXX)
enable_testing()
if(NOT DEFINED PRIVATE_ROOT OR NOT DEFINED SALTS_INSTALLED_ROOT)
  message(FATAL_ERROR "Private staged root and installed Salts SDK are required")
endif()
if(NOT EXISTS "${PRIVATE_ROOT}/include/salts/service_host.h" OR
   NOT EXISTS "${PRIVATE_ROOT}/include/salts/configurator.h")
  message(FATAL_ERROR "Private candidate headers missing from staged root")
endif()
if(EXISTS "${SALTS_INSTALLED_ROOT}/include/salts/service_host.h")
  message(FATAL_ERROR "Published Salts SDK unexpectedly exports draft ServiceHost")
endif()
find_package(Salts CONFIG REQUIRED PATHS "${SALTS_INSTALLED_ROOT}" NO_DEFAULT_PATH)
if(TARGET Salts::ServiceHostCandidate OR TARGET Salts::ConfiguratorCandidate)
  message(FATAL_ERROR "Draft ServiceHost leaked into public installed Salts targets")
endif()
foreach(required IN ITEMS Salts::ComponentABI Salts::ComponentPluginABI Salts::Plugin Salts::Platform)
  if(NOT TARGET ${required})
    message(FATAL_ERROR "Required installed dependency target missing: ${required}")
  endif()
endforeach()
add_library(staged_configurator STATIC IMPORTED GLOBAL)
set_target_properties(staged_configurator PROPERTIES
  IMPORTED_LOCATION "${PRIVATE_ROOT}/lib/${CONFIGURATOR_FILENAME}"
  INTERFACE_INCLUDE_DIRECTORIES "${PRIVATE_ROOT}/include"
  INTERFACE_LINK_LIBRARIES "Salts::ComponentABI")
add_library(staged_service_host STATIC IMPORTED GLOBAL)
set_target_properties(staged_service_host PROPERTIES
  IMPORTED_LOCATION "${PRIVATE_ROOT}/lib/${HOST_FILENAME}"
  INTERFACE_INCLUDE_DIRECTORIES "${PRIVATE_ROOT}/include"
  INTERFACE_LINK_LIBRARIES "staged_configurator;Salts::ComponentPluginABI;Salts::Plugin;Salts::Platform")
if(PRIVATE_ASAN OR PRIVATE_UBSAN)
  if(NOT CMAKE_C_COMPILER_ID MATCHES "Clang|GNU")
    message(FATAL_ERROR "Sanitizer consumer requires GCC/Clang")
  endif()
  if(PRIVATE_ASAN)
    add_compile_options(-fsanitize=address -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address)
  endif()
  if(PRIVATE_UBSAN)
    add_compile_options(-fsanitize=undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=undefined)
  endif()
endif()
foreach(lang IN ITEMS c cpp)
  if(lang STREQUAL "c")
    add_executable(staged_host_consumer_c consumer_c.c)
    set_target_properties(staged_host_consumer_c PROPERTIES
      C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
    set(consumer staged_host_consumer_c)
  else()
    add_executable(staged_host_consumer_cpp consumer_cpp.cpp)
    set_target_properties(staged_host_consumer_cpp PROPERTIES
      CXX_STANDARD 17 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
    set(consumer staged_host_consumer_cpp)
  endif()
  target_link_libraries(${consumer} PRIVATE staged_service_host)
  add_test(NAME ${consumer} COMMAND ${consumer}
    "${PRIVATE_ROOT}/plugin/${PROVIDER1_FILENAME}"
    "${PRIVATE_ROOT}/plugin/${PROVIDER2_FILENAME}")
endforeach()
message(STATUS "Private staged SDK only: no public Salts candidate export or install")
